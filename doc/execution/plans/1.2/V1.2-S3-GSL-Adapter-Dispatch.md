# V1.2-S3: GSL Adapter Dispatch — Register GSL Statistical Functions

**Goal:** Register GSL (GNU Scientific Library) statistical functions behind the capability registry, providing a second backend alongside Arrow.

**Architecture:** A GSL adapter module that maps Yamori capability descriptors to GSL functions. The adapter registers with the capability registry. Since GSL is a C library, this requires `extern "c"` bindings and C source compilation in the Zig build. GSL functions are registered as capabilities with `.backend_selector = .gsl`.

**Tech Stack:** Zig 0.17, GSL C bindings, `extern "c"`, build.zig C source compilation

**Step 1: Write the GSLFunctionMap struct**

```zig
pub const GSLFunctionMap = struct {
    const Entry = struct {
        capability_name: []const u8,
        gsl_fn: []const u8,
        gsl_header: []const u8,
    };

    pub const all: []const Entry = &.{
        .{ .capability_name = "mean", .gsl_fn = "gsl_stats_mean", .gsl_header = "mean" },
        .{ .capability_name = "variance", .gsl_fn = "gsl_stats_variance", .gsl_header = "variance" },
        .{ .capability_name = "std_dev", .gsl_fn = "gsl_stats_sd", .gsl_header = "sd" },
        .{ .capability_name = "covariance", .gsl_fn = "gsl_stats_covariance", .gsl_header = "covariance" },
        .{ .capability_name = "correlation", .gsl_fn = "gsl_stats_correlation", .gsl_header = "correlation" },
        .{ .capability_name = "quantile", .gsl_fn = "gsl_stats_quantile", .gsl_header = "quantile" },
        .{ .capability_name = "median", .gsl_fn = "gsl_stats_median_from_sorted_data", .gsl_header = "median" },
        .{ .capability_name = "percentile", .gsl_fn = "gsl_stats_percentile_from_sorted_data", .gsl_header = "percentile" },
    };

    pub fn lookup(comptime capability_name: []const u8) ?[]const u8 {
        inline for (all) |entry| {
            if (std.mem.eql(u8, entry.capability_name, capability_name)) {
                return entry.gsl_fn;
            }
        }
        return null;
    }
};
```

## Task 3: Implement GSL Function Registry

**Objective:** Create the registry that maps capability names to GSL function metadata.

**Files:**
- Modify: `src/gsl_adapter.zig`

**Step 1: Write the GSLFunctionRegistry struct**

```zig
pub const GSLFunctionRegistry = struct {
    allocator: std.mem.Allocator,
    functions: std.StringHashMap(GSLFunctionEntry),

    pub const GSLFunctionEntry = struct {
        name: []const u8,
        gsl_fn_name: []const u8,
        gsl_header: []const u8,
        min_inputs: usize,
        max_inputs: usize,
    };

    pub fn init(gpa: std.mem.Allocator) GSLFunctionRegistry {
        return GSLFunctionRegistry{
            .allocator = gpa,
            .functions = std.StringHashMap(GSLFunctionEntry).init(gpa),
        };
    }

    pub fn deinit(self: *GSLFunctionRegistry) void {
        var it = self.functions.keyIterator();
        while (it.next()) |key| {
            var fn_entry = self.functions.getPtr(key.*) orelse unreachable;
            self.allocator.free(fn_entry.name);
            self.allocator.free(fn_entry.gsl_fn_name);
            self.allocator.free(fn_entry.gsl_header);
        }
        self.functions.deinit();
    }

    pub fn register(self: *GSLFunctionRegistry, entry: GSLFunctionEntry) !void {
        try self.functions.put(entry.name, entry);
    }

    pub fn get(self: *const GSLFunctionRegistry, name: []const u8) ?*const GSLFunctionEntry {
        return self.functions.get(name);
    }
};
```

**Files:**

**Step 1: Add input validation helper**

```zig
pub fn validateGSLInput(inputs: []const []const f64) GSLAdapterError!void {
    if (inputs.len == 0) return GSLAdapterError.InvalidInput;
    
    const first_len = inputs[0].len;
    if (first_len == 0) return GSLAdapterError.InvalidInput;
    
    var i: usize = 1;
    while (i < inputs.len) : (i += 1) {
        if (inputs[i].len != first_len) {
            return GSLAdapterError.InvalidInput;
        }
    }
}
```

## Task 6: Test GSL and Arrow Capability Coexistence

**Objective:** Verify that GSL and Arrow capabilities can coexist in the same registry without conflicts.

**Files:**
- Test: `src/main.zig` (append test block)

**Step 1: Write coexistence test**

```zig
test "GSL and Arrow capabilities coexist in CapabilityRegistry" {
    var gpa = std.testing.allocator;
    var cap_reg = CapabilityRegistry.init(gpa);
    defer cap_reg.deinit();

    // Register Arrow capability
    const arrow_cap = Capability.init(
        gpa, "add", 1, "Arrow add",
        &.{Capability.TypeDescriptor{ .category = .vector, .element_type = .f64 }},
        .{},
        .arrow,
    ) catch |err| return err;
    defer arrow_cap.deinit(gpa);
    try cap_reg.register(arrow_cap);

    // Register GSL capability
    const gsl_cap = Capability.init(
        gpa, "mean", 1, "GSL mean",
        &.{Capability.TypeDescriptor{ .category = .vector, .element_type = .f64 }},
        .{},
        .gsl,
    ) catch |err| return err;
    defer gsl_cap.deinit(gpa);
    try cap_reg.register(gsl_cap);

    try std.testing.expectEqual(@as(usize, 2), cap_reg.capabilities.count());
    try std.testing.expectEqualStrings("arrow", @tagName(cap_reg.get("add").?.backend_selector));
    try std.testing.expectEqualStrings("gsl", @tagName(cap_reg.get("mean").?.backend_selector));
}
```
