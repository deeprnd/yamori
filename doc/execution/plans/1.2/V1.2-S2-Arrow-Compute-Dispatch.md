# V1.2-S2: Arrow Compute Dispatch — Register Arrow Compute Functions

**Goal:** Register Arrow Compute functions behind the capability registry, enabling dispatch to Apache Arrow's compute layer.

**Architecture:** An Arrow adapter layer that maps Yamori capability descriptors to Arrow Compute functions. The adapter registers itself with the capability registry at initialization. Each Arrow Compute function is wrapped with type-checking and input/output validation.

**Tech Stack:** Zig 0.17, Arrow C Data Interface (extern "c" bindings), `std.ArrayList`

**Step 1: Write the ArrowComputeMap struct**

```zig
pub const ArrowComputeMap = struct {
    // Maps Yamori capability names to Arrow Compute function names
    const Entry = struct {
        capability_name: []const u8,
        arrow_fn: []const u8,
    };

    pub const all: []const Entry = &.{
        .{ .capability_name = "add", .arrow_fn = "add" },
        .{ .capability_name = "subtract", .arrow_fn = "subtract" },
        .{ .capability_name = "multiply", .arrow_fn = "multiply" },
        .{ .capability_name = "divide", .arrow_fn = "divide" },
        .{ .capability_name = "sum", .arrow_fn = "sum" },
        .{ .capability_name = "mean", .arrow_fn = "mean" },
        .{ .capability_name = "min", .arrow_fn = "min" },
        .{ .capability_name = "max", .arrow_fn = "max" },
        .{ .capability_name = "count", .arrow_fn = "count" },
        .{ .capability_name = "std_dev", .arrow_fn = "stddev" },
        .{ .capability_name = "variance", .arrow_fn = "variance" },
    };

    pub fn lookup(comptime capability_name: []const u8) ?[]const u8 {
        inline for (all) |entry| {
            if (std.mem.eql(u8, entry.capability_name, capability_name)) {
                return entry.arrow_fn;
            }
        }
        return null;
    }
};
```

## Task 3: Implement Arrow Compute Function Registry

**Objective:** Create the registry that maps capability names to Arrow Compute function pointers/wrappers.

**Files:**
- Modify: `src/arrow_adapter.zig`

**Step 1: Write the ArrowFunctionRegistry struct**

```zig
pub const ArrowFunctionRegistry = struct {
    allocator: std.mem.Allocator,
    functions: std.StringHashMap(*const ArrowComputeFunction),

    pub const ArrowComputeFunction = struct {
        name: []const u8,
        // In S2, we store the Arrow Compute function name string.
        // Actual dispatch implementation is deferred to S5 (backend-agnostic API).
        arrow_fn_name: []const u8,
    };

    pub fn init(gpa: std.mem.Allocator) ArrowFunctionRegistry {
        return ArrowFunctionRegistry{
            .allocator = gpa,
            .functions = std.StringHashMap(*const ArrowComputeFunction).init(gpa),
        };
    }

    pub fn deinit(self: *ArrowFunctionRegistry) void {
        self.functions.deinit();
    }

    pub fn register(self: *ArrowFunctionRegistry, fn_entry: ArrowComputeFunction) !void {
        const duped_name = try self.allocator.dupe(u8, fn_entry.name);
        errdefer self.allocator.free(duped_name);
        const duped_arrow = try self.allocator.dupe(u8, fn_entry.arrow_fn_name);
        errdefer self.allocator.free(duped_arrow);

        // Allocate a const struct — we own the backing memory via the hash map
        // For now, store a pointer to a heap-allocated copy
        const stored = try self.allocator.create(ArrowComputeFunction);
        stored.name = duped_name;
        stored.arrow_fn_name = duped_arrow;
        try self.functions.put(stored.name, stored);
    }

    pub fn get(self: *const ArrowFunctionRegistry, name: []const u8) ?*const ArrowComputeFunction {
        return self.functions.get(name);
    }
};
```

```zig
test "ArrowFunctionRegistry populates from CapabilityRegistry" {
    var gpa = std.testing.allocator;
    var cap_reg = CapabilityRegistry.init(gpa);
    defer cap_reg.deinit();

    var arrow_reg = ArrowFunctionRegistry.init(gpa);
    defer arrow_reg.deinit();

    // Create capabilities that match ArrowComputeMap entries
    inline for (ArrowComputeMap.all) |entry| {
        const cap = Capability.init(
            gpa,
            entry.capability_name,
            1,
            "Arrow-implemented " ++ entry.capability_name,
            &.{Capability.TypeDescriptor{ .category = .vector, .element_type = .f64 }},
            Capability.TypeDescriptor{ .category = .scalar, .element_type = .f64 },
            .arrow,
        ) catch |err| return err;
        defer cap.deinit(gpa);
        try cap_reg.register(cap);

        const arrow_fn = ArrowFunctionRegistry.ArrowComputeFunction{
            .name = entry.capability_name,
            .arrow_fn_name = entry.arrow_fn,
        };
        try arrow_reg.register(arrow_fn);
    }

    // Verify all capabilities have matching Arrow functions
    var it = cap_reg.capabilities.iterator();
    while (it.next()) |entry| |cap| {
        if (cap.backend_selector != .arrow) continue;
        const arrow_fn = arrow_reg.get(cap.name) orelse {
            try std.testing.expect(false); // Should not happen
            return;
        };
        const expected = ArrowComputeMap.lookup(cap.name) orelse {
            try std.testing.expect(false);
            return;
        };
        try std.testing.expectEqualStrings(expected, arrow_fn.arrow_fn_name);
    }
}
```

```zig
pub fn validateInputLengths(inputs: []const []const f64) ArrowError!void {
    if (inputs.len == 0) return ArrowError.NullInput;
    
    const first_len = inputs[0].len;
    if (first_len == 0) return ArrowError.InvalidInputLength;
    
    var i: usize = 1;
    while (i < inputs.len) : (i += 1) {
        if (inputs[i].len != first_len) {
            return ArrowError.InvalidInputLength;
        }
    }
}
```
