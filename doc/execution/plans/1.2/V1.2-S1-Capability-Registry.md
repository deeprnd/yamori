# V1.2-S1: Capability Registry — Schema and Storage

**Goal:** Define the capability registry schema and storage mechanism for mapping operations to backend implementations.

**Architecture:** A central registry struct that stores operation-to-backend mappings with versioned capability descriptors. Each capability has an operation name, accepted input types, output types, and backend selector. Registry is immutable after build-time registration, mutable only during the test harness via explicit registration calls.

**Tech Stack:** Zig 0.17 stdlib, `std.mem.Allocator`, `std.StringHashMap`, `std.json`

**Objective:** Create the Capability struct that represents a single registered capability.

**Files:**

**Step 1: Write the Capability struct definition**

```zig
pub const Capability = struct {
    name: []const u8,
    version: u32,
    description: []const u8,
    input_types: []const TypeDescriptor,
    output_type: TypeDescriptor,
    backend_selector: BackendSelector,

    pub const TypeDescriptor = struct {
        category: CapabilityCategory,
        element_type: ?ElementType,
    };

    pub const CapabilityCategory = enum { scalar, vector, matrix, tensor };
    pub const ElementType = enum { f32, f64, i32, i64, u32, u64, bool, string };

    pub const BackendSelector = enum { arrow, gsl, default };

    pub fn init(
        gpa: std.mem.Allocator,
        name: []const u8,
        version: u32,
        description: []const u8,
        input_types: []const TypeDescriptor,
        output_type: TypeDescriptor,
        selector: BackendSelector,
    ) !Capability {
        const inputs = try gpa.dupe(TypeDescriptor, input_types);
        errdefer gpa.free(inputs);
        return Capability{
            .name = name,
            .version = version,
            .description = description,
            .input_types = inputs,
            .output_type = output_type,
            .backend_selector = selector,
        };
    }

    pub fn deinit(self: *Capability, gpa: std.mem.Allocator) void {
        gpa.free(self.input_types);
        self.input_types = &.{};
    }
};
```

## Task 2: Define the Registry Schema

**Objective:** Create the CapabilityRegistry struct with registration and lookup methods.

**Files:**
- Create: `src/registry.zig`
- Test: `src/main.zig` (append test block)

**Step 1: Write the CapabilityRegistry struct**

```zig
pub const CapabilityRegistry = struct {
    allocator: std.mem.Allocator,
    capabilities: std.StringHashMap(Capability),

    pub fn init(gpa: std.mem.Allocator) CapabilityRegistry {
        return CapabilityRegistry{
            .allocator = gpa,
            .capabilities = std.StringHashMap(Capability).init(gpa),
        };
    }

    pub fn deinit(self: *CapabilityRegistry) void {
        var it = self.capabilities.keyIterator();
        while (it.next()) |key| {
            var cap = self.capabilities.getPtr(key.*) orelse unreachable;
            cap.deinit(self.allocator);
        }
        self.capabilities.deinit();
    }

    pub fn register(self: *CapabilityRegistry, cap: Capability) !void {
        try self.capabilities.put(cap.name, cap);
    }

    pub fn get(self: *const CapabilityRegistry, name: []const u8) ?*const Capability {
        return self.capabilities.getConst(name);
    }

    pub fn contains(self: *const CapabilityRegistry, name: []const u8) bool {
        return self.capabilities.contains(name);
    }
};
```

```zig
pub fn toJson(self: *const CapabilityRegistry, gpa: std.mem.Allocator) ![]u8 {
    var buf = try std.ArrayList(u8).initCapacity(gpa, 0);
    errdefer buf.deinit(gpa);

    try buf.appendSlice(gpa, "{\"capabilities\":[");

    var first = true;
    var it = self.capabilities.iterator();
    while (it.next()) |entry| {
        if (!first) try buf.appendSlice(gpa, ",");
        first = false;
        const ver_buf = blk: {
            var tmp: [20]u8 = undefined;
            break :blk std.fmt.bufPrint(&tmp, "{d}", .{entry.value_ptr.version}) catch "0";
        };
        const tag = @tagName(entry.value_ptr.backend_selector);
        try buf.appendSlice(gpa, "{\"name\":\"");
        try buf.appendSlice(gpa, entry.value_ptr.name);
        try buf.appendSlice(gpa, "\",\"version\":");
        try buf.appendSlice(gpa, ver_buf);
        try buf.appendSlice(gpa, ",\"description\":\"");
        try buf.appendSlice(gpa, entry.value_ptr.description);
        try buf.appendSlice(gpa, "\",\"backend_selector\":\"");
        try buf.appendSlice(gpa, tag);
        try buf.appendSlice(gpa, "\"}");
    }

    try buf.appendSlice(gpa, "]}");
    return buf.toOwnedSlice(gpa);
}
```
