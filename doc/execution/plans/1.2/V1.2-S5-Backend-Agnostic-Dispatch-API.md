# V1.2-S5: Backend-Agnostic Dispatch API — Domain Code Calls dispatch(op, inputs)

**Goal:** Create a backend-agnostic dispatch API where domain code calls `dispatch(op, inputs)` without knowing which backend implements the operation.

**Architecture:** A single `dispatch` function that receives an operation name and input data, looks up the capability in the registry, selects the appropriate backend (Arrow or GSL), and calls the backend's compute function. The API is unified — callers never see Arrow or GSL types.

**Tech Stack:** Zig 0.17, `std.mem.Allocator`, `std.ArrayList`, capability registry integration

## Task 1: Define the Dispatch API Signature

**Objective:** Create the dispatch function signature and result type.

**Files:**

**Step 1: Write the dispatch API**

```zig
pub const DispatchResult = struct {
    data: []f64,
    count: usize,

    pub fn deinit(self: *DispatchResult, gpa: std.mem.Allocator) void {
        if (self.data.len > 0) {
            gpa.free(self.data);
            self.data = &.{};
        }
        self.count = 0;
    }
};

pub const DispatchError = error{
    UnknownOperation,
    BackendNotAvailable,
    BackendComputeFailed,
    InvalidOperationInput,
    InternalError,
} || YamoriError;

pub const Dispatcher = struct {
    cap_reg: *const CapabilityRegistry,
    arrow_reg: *const ArrowFunctionRegistry,
    gsl_reg: *const GSLFunctionRegistry,

    pub fn init(
        cap_reg: *const CapabilityRegistry,
        arrow_reg: *const ArrowFunctionRegistry,
        gsl_reg: *const GSLFunctionRegistry,
    ) Dispatcher {
        return Dispatcher{
            .cap_reg = cap_reg,
            .arrow_reg = arrow_reg,
            .gsl_reg = gsl_reg,
        };
    }

    /// Dispatch an operation to its registered backend.
    /// Caller is responsible for deinit(result) on success.
    pub fn dispatch(
        self: Dispatcher,
        gpa: std.mem.Allocator,
        op_name: []const u8,
        inputs: []const []const f64,
    ) DispatchError!DispatchResult {
        // 1. Look up capability in registry
        const cap = self.cap_reg.get(op_name) orelse {
            return YamoriError.UnknownOperation;
        };

        // 2. Validate inputs
        try ArrowAdapter.validateInputLengths(inputs);

        // 3. Route to appropriate backend
        return switch (cap.backend_selector) {
            .arrow => dispatchArrow(self, gpa, op_name, inputs),
            .gsl => dispatchGSL(self, gpa, op_name, inputs),
            .default => dispatchArrow(self, gpa, op_name, inputs),
        };
    }
};
```

## Task 2: Implement Arrow Dispatch Path

**Objective:** Implement the Arrow backend dispatch path.

**Files:**
- Modify: `src/dispatch.zig`

**Step 1: Add Arrow dispatch helper**

```zig
fn dispatchArrow(
    self: Dispatcher,
    gpa: std.mem.Allocator,
    op_name: []const u8,
    inputs: []const []const f64,
) DispatchError!DispatchResult {
    const arrow_fn = self.arrow_reg.get(op_name) orelse {
        return YamoriError.BackendNotAvailable;
    };

    const arrow_fn_name = arrow_fn.arrow_fn_name;

    // In S5, we mock the Arrow compute call.
    // Actual Arrow C Data Interface dispatch is deferred to S5 implementation.
    // For now, return a simple transformation result based on the operation.
    
    // Validate inputs first
    try ArrowAdapter.validateInputLengths(inputs);
    
    const len = inputs[0].len;
    const output = try gpa.alloc(f64, len);
    errdefer gpa.free(output);

    // Mock compute: for testing, return the first input array
    @memcpy(output, inputs[0]);

    return DispatchResult{
        .data = output,
        .count = len,
    };
}
```

```zig
fn dispatchGSL(
    self: Dispatcher,
    gpa: std.mem.Allocator,
    op_name: []const u8,
    inputs: []const []const f64,
) DispatchError!DispatchResult {
    const gsl_fn = self.gsl_reg.get(op_name) orelse {
        return YamoriError.BackendNotAvailable;
    };

    try GSLAdapter.validateGSLInput(inputs);
    
    const len = inputs[0].len;
    const output = try gpa.alloc(f64, 1); // GSL functions typically return scalar
    errdefer gpa.free(output);

    // Mock compute: return a placeholder value
    output[0] = 0.0;

    return DispatchResult{
        .data = output,
        .count = 1,
    };
}
```
