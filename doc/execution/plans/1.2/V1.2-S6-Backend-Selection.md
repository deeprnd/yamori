# V1.2-S6: Backend Selection — Runtime and Compile-Time Selection of Backend Per Operation

**Goal:** Implement runtime and compile-time backend selection, allowing the caller to override the default backend per operation.

**Architecture:** A backend policy layer that supports both compile-time defaults (from the capability registry) and runtime overrides. The selection mechanism checks runtime policy first, then falls back to the capability's default selector. This enables testing with different backends without modifying capability definitions.

**Tech Stack:** Zig 0.17, `std.StringHashMap`, compile-time options, runtime configuration

## Task 1: Define Backend Policy Types

**Objective:** Create the backend policy structs for compile-time and runtime selection.

**Files:**
- Create: `src/backend_policy.zig`
- Test: `src/main.zig` (append test block)

**Step 1: Write backend policy types**

```zig
pub const BackendPolicy = enum { arrow, gsl, auto };

pub const BackendSelector = struct {
    runtime_overrides: std.StringHashMap(BackendPolicy),
    compile_time_default: BackendPolicy,

    pub fn init(gpa: std.mem.Allocator, default: BackendPolicy) BackendSelector {
        return BackendSelector{
            .runtime_overrides = std.StringHashMap(BackendPolicy).init(gpa),
            .compile_time_default = default,
        };
    }

    pub fn deinit(self: *BackendSelector) void {
        self.runtime_overrides.deinit();
    }

    /// Set runtime override for a specific operation
    pub fn setOverride(self: *BackendSelector, op_name: []const u8, policy: BackendPolicy) !void {
        try self.runtime_overrides.put(op_name, policy);
    }

    /// Clear runtime override for a specific operation
    pub fn clearOverride(self: *BackendSelector, op_name: []const u8) void {
        self.runtime_overrides.remove(op_name);
    }

    /// Select backend for an operation, checking runtime overrides first
    pub fn select(self: *const BackendSelector, op_name: []const u8, default_policy: BackendPolicy) BackendPolicy {
        if (self.runtime_overrides.get(op_name)) |override| {
            return override;
        }
        return default_policy;
    }
};
```

## Task 3: Integrate BackendSelector with Dispatcher

**Objective:** Add BackendSelector to the Dispatcher struct and update dispatch() to use it.

**Files:**
- Modify: `src/dispatch.zig`

**Step 1: Add BackendSelector to Dispatcher**

```zig
pub const Dispatcher = struct {
    cap_reg: *const CapabilityRegistry,
    arrow_reg: *const ArrowFunctionRegistry,
    gsl_reg: *const GSLFunctionRegistry,
    backend_selector: *const BackendSelector,  // NEW

    pub fn init(
        cap_reg: *const CapabilityRegistry,
        arrow_reg: *const ArrowFunctionRegistry,
        gsl_reg: *const GSLFunctionRegistry,
        backend_selector: *const BackendSelector,
    ) Dispatcher {
        return Dispatcher{
            .cap_reg = cap_reg,
            .arrow_reg = arrow_reg,
            .gsl_reg = gsl_reg,
            .backend_selector = backend_selector,
        };
    }

    pub fn dispatch(
        self: Dispatcher,
        gpa: std.mem.Allocator,
        op_name: []const u8,
        inputs: []const []const f64,
    ) DispatchError!DispatchResult {
        const cap = self.cap_reg.get(op_name) orelse {
            return YamoriError.UnknownOperation;
        };

        try ArrowAdapter.validateInputLengths(inputs);

        // Use backend selector instead of cap.backend_selector directly
        const policy = self.backend_selector.select(
            op_name,
            cap.backend_selector,
        );

        return switch (policy) {
            .arrow => dispatchArrow(self, gpa, op_name, inputs),
            .gsl => dispatchGSL(self, gpa, op_name, inputs),
            .auto => switch (cap.backend_selector) {
                .arrow => dispatchArrow(self, gpa, op_name, inputs),
                .gsl => dispatchGSL(self, gpa, op_name, inputs),
                .default => dispatchArrow(self, gpa, op_name, inputs),
            },
        };
    }
};
```

test "Dispatcher.backend_selector can switch backends at runtime" {
    var gpa = std.testing.allocator;
    var cap_reg = CapabilityRegistry.init(gpa);
    defer cap_reg.deinit();

    var arrow_reg = ArrowFunctionRegistry.init(gpa);
    defer arrow_reg.deinit();

    var gsl_reg = GSLFunctionRegistry.init(gpa);
    defer gsl_reg.deinit();

    var selector = BackendSelector.init(gpa, .arrow);
    defer selector.deinit();

    const cap = Capability.init(
        gpa, "add", 1, "Arrow add",
        &.{Capability.TypeDescriptor{ .category = .vector, .element_type = .f64 }},
        .{},
        .arrow,
    ) catch |err| return err;
    defer cap.deinit(gpa);
    try cap_reg.register(cap);

    const arrow_fn = ArrowFunctionRegistry.ArrowComputeFunction{
        .name = "add",
        .arrow_fn_name = "add",
    };
    try arrow_reg.register(arrow_fn);

    const dispatcher = Dispatcher.init(&cap_reg, &arrow_reg, &gsl_reg, &selector);

    const input = [3]f64{ 1.0, 2.0, 3.0 };

    // Dispatch with Arrow (default)
    var result = dispatcher.dispatch(gpa, "add", &.{ &input }) catch |err| return err;
    defer result.deinit(gpa);
    try std.testing.expectEqual(@as(usize, 3), result.count);

    // Switch to GSL
    selector.setOverride("add", .gsl);
    result = dispatcher.dispatch(gpa, "add", &.{ &input }) catch |err| return err;
    defer result.deinit(gpa);
    try std.testing.expectEqual(@as(usize, 1), result.count);

    // Switch back to Arrow
    selector.setOverride("add", .arrow);
    result = dispatcher.dispatch(gpa, "add", &.{ &input }) catch |err| return err;
    defer result.deinit(gpa);
    try std.testing.expectEqual(@as(usize, 3), result.count);
}
